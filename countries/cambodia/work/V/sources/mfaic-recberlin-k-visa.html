ទិដ្ឋាការពិសេស ឬ ទិដ្ឋាការខ្មែរ (K-Visa)

ទិដ្ឋាការពិសេស ឬ ទិដ្ឋាការខ្មែរ (K-Visa) គឺជាទិដ្ឋាការសម្រាប់ពលរដ្ឋដែលមានឈាមជ័រជាខ្មែរដែលប្រើប្រាស់លិខិតឆ្លងដែនបរទេសសម្រាប់ធ្វើដំណើរ ក្នុងនោះអាចជាពលរដ្ឋខ្មែរដែលមានសញ្ជាតិពីរ ឬ អ្នកដែលមានឪពុកម្តាយទាំងសងខាង ឬ តែម្ខាងមានដើមកំណើតខ្មែរ។ ដោយសារតែលោកអ្នកប្រើប្រាស់លិខិតឆ្លងដែនបរទេសសម្រាប់ធ្វើដំណើរ លោកអ្នកពុំអាចធ្វើដំណើរទៅប្រទេសកម្ពុជាបាននោះទេ គឺតម្រូវឱ្យមានទិដ្ឋាការពិសេស ឬ ទិដ្ឋាការខ្មែរ នេះជាមុនសិន។

លក្ខណៈពិសេស

​ជាប្រភេទទិដ្ឋាការដែល៖

  • អាចចេញចូលច្រើនដង (Multiple Entries)។
  • មិនផុតកំណត់ (លើកលែងតែ​លិខិតឆ្លងដែន​អស់សុពលភាព ខូច ឬ​ បាត់)។
  • អនុញ្ញាតឱ្យស្នាក់នៅក្នុងប្រទេសកម្ពុជាដោយមិនកំណត់រយៈពេល។
  • ឥតគិតថ្លៃ។

ការផ្លាស់ប្តូរសំខាន់ៗ​ដែលលោកអ្នកត្រូវដឹង៖ ចាប់ពី ថ្ងៃទី៣ ខែតុលា ឆ្នាំ២០២៦ តទៅ ការផ្តល់ទិដ្ឋាការប្រភេទ K ត្រូវបានផ្លាស់ប្តូរទាំងស្រុងពីការបិទសន្លឹកទិដ្ឋាការស្អិតលើលិខិតឆ្លងដែន មកជាទម្រង់ទិដ្ឋាការអេឡិចត្រូនិកវិញ។ ទិដ្ឋាការស្អិត​ដែលបាន​ចេញជូនមុនថ្ងៃទី៣ ខែតុលា ឆ្នាំ២០២៦ នៅមានសុពលភាពប្រើប្រាស់ធម្មតា។ បច្ចុប្បុន្ននេះ សុំទិដ្ឋាការ K អាចស្នើសុំដោយផ្ទាល់​បានតាមរយៈថ្នាល​អេឡិចត្រូនិក (e-Visa portal) ដែលមានអាសយដ្ឋាន evisa.gov.kh ឬ សុំតាមរយៈស្ថានទូត (ផ្ញើអ៊ីម៉ែល) ដោយបំពេញពាក្យស្នើសុំដែលអាចទាញយកបាននៅ ទីនេះ។

កំណត់សម្គាល់៖

■ ការស្នើសុំ និងការផ្តល់ទិដ្ឋាការប្រភេទ K គឺធ្វើឡើងតាមប្រព័ន្ធអេឡិចត្រូនិកទាំងស្រុង ដោយពុំចាំបាច់​នាំយក ឬ ផ្ញើ​លិខិត​ឆ្លងដែន​មកស្ថានទូតទៀត​នោះទេ។

■ គ្រប់​ប្រភេទទិដ្ឋាការទាំងអស់ រួមទាំងទិដ្ឋាการប្រភេទ K ផងដែរ នឹងត្រូវបាន​សម្រេចផ្តល់​ជូន ឬ បដិសេធ​ដោយផ្ទាល់ពីការិយាល័យទិដ្ឋាការអេឡិចត្រូនិក ដោយស្ថានទូតពុំមែនជាអ្នកសម្រេចនោះឡើយ។

■ ជនបរទេសដែលរៀបការជាមួយពលរដ្ឋខ្មែរ ពុំទទួលបានសញ្ជាតិខ្មែរដោយស្វ័យប្រវត្តិនោះទេ ដែលមានន័យថា ពួកគេមិនអាចស្នើសុំទិដ្ឋាការខ្មែរនេះបាននោះដែរ គឺពួកគេត្រូវការទិដ្ឋាការទេសចរ ឬ ទិដ្ឋាការធម្មតា (ប្រភេទ T ឬ E).

រាល់ចម្ងល់ផ្សេងៗ លោកអ្នកអាចទូរសព្ទសាកសួរមកកាន់ស្ថានទូតនៅរៀងរាល់ម៉ោងធ្វើការ។
